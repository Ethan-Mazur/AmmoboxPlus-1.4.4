using Terraria.ID;
using Terraria.ModLoader;

namespace AmmoboxPlus.Items.Weapons {
    class ammoClassName : ModItem {
        public override void SetStaticDefaults() {
            DisplayName.SetDefault("ammoIngameName");
            Tooltip.SetDefault("ammoDescription");
        }

        public override void SetDefaults() {
            item.damage = ammoDamage;
            item.ranged = true;
            item.width = 20;
            item.height = 14;
            item.maxStack = 999;
            item.consumable = true;
            item.knockBack = 2f;
            item.value = 15;
            item.rare = ItemRarityID.Lime;
            item.shootSpeed = ammoVelocityf;
            item.shoot = mod.ProjectileType("ammoClassName");
            item.ammo = AmmoID.Rocket;
        }

        public override void AddRecipes() {
            ModRecipe recipe = new ModRecipe(mod);
            recipe.AddIngredient(ammoFirstIngredient, ammoFirstIngredientAmount);
            recipe.AddIngredient(ammoSecondIngredient, ammoSecondIngredientAmount);
            recipe.AddTile(TileID.MythrilAnvil);
            recipe.SetResult(this, ammoCraftCount);
            recipe.AddRecipe();
        }

    }
}
